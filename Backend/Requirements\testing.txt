import numpy as np
import pandas as pd
import pickle

#Загрузка модели с файла + артефакты и скейлер
def load_model(path='credit_model_complete.pkl'):
    with open(path, 'rb') as f:
        artifacts = pickle.load(f)
    return artifacts

artifacts = load_model()
model = artifacts['model']
scaler = artifacts['scaler']
onehot_encoder = artifacts['onehot_encoder']
mlb = artifacts['mlb']
feature_names = artifacts['feature_names']
label_encoder_target = artifacts['label_encoder_target']

#Предобработка
def preprocess_new_customer(raw_data):
    """
    raw_data: словарь с ключами которые обязательны для ии при предсказывании:
        age, annual_income, monthly_inhand_salary, credit_history_age,
        total_emi_per_month, num_bank_accounts, num_credit_card,
        interest_rate, num_of_loan, delay_from_due_date,
        num_of_delayed_payment, changed_credit_limit, num_credit_inquiries,
        outstanding_debt, credit_utilization_ratio, amount_invested_monthly,
        monthly_balance, occupation, credit_mix, payment_of_min_amount,
        payment_behaviour, type_of_loan (list of strings)
    """
    #Нвый дф для пользователя (в конце можно сделать своего)
    df_new = pd.DataFrame([raw_data])
    
    #Обработка списка
    if 'type_of_loan' in df_new.columns:

        def parse_list(x):
            if isinstance(x, str):
                return x.strip("[]").replace("'", "").replace('"', '').split(',')
            return x if isinstance(x, list) else []
        df_new['type_of_loan_parsed'] = df_new['type_of_loan'].apply(parse_list)

        loan_dummies = pd.DataFrame(
            mlb.transform(df_new['type_of_loan_parsed']),
            columns=[f'loan_{c}' for c in mlb.classes_],
            index=df_new.index
        )
        df_new = pd.concat([df_new.drop(columns=['type_of_loan', 'type_of_loan_parsed'], errors='ignore'), loan_dummies], axis=1)
    else:

        for col in [f'loan_{c}' for c in mlb.classes_]:
            df_new[col] = 0
    
    #Значения в флоат + очистка от минусов( - )
    numeric_cols = ['age', 'annual_income', 'monthly_inhand_salary', 'credit_history_age',
                    'total_emi_per_month', 'num_bank_accounts', 'num_credit_card',
                    'interest_rate', 'num_of_loan', 'delay_from_due_date',
                    'num_of_delayed_payment', 'changed_credit_limit', 'num_credit_inquiries',
                    'outstanding_debt', 'credit_utilization_ratio', 'amount_invested_monthly',
                    'monthly_balance']
    for col in numeric_cols:
        if col in df_new.columns:
            #Запятые убираем
            df_new[col] = df_new[col].astype(str).str.replace(',', '').str.replace('%', '').str.strip()
            df_new[col] = pd.to_numeric(df_new[col], errors='coerce')
    
    # Ванхот
    cat_cols = ['occupation', 'credit_mix', 'payment_of_min_amount', 'payment_behaviour']
    cat_encoded = onehot_encoder.transform(df_new[cat_cols])
    cat_names = onehot_encoder.get_feature_names_out(cat_cols)
    cat_df = pd.DataFrame(cat_encoded, columns=cat_names, index=df_new.index)
    
    df_new = df_new.drop(columns=cat_cols, errors='ignore')
    df_new = pd.concat([df_new, cat_df], axis=1)
    
    #Проверка признаков
    for col in feature_names:
        if col not in df_new.columns:
            df_new[col] = 0

    df_new = df_new[feature_names]
    
    #Масштабирование
    X_scaled = scaler.transform(df_new)
    return X_scaled


def predict_credit_score(raw_data):
    X_processed = preprocess_new_customer(raw_data)
    pred_class = model.predict(X_processed)[0]
    
    if label_encoder_target is not None:
        pred_label = label_encoder_target.inverse_transform([pred_class])[0]
    else:
        pred_label = pred_class
    
    return pred_class, pred_label
